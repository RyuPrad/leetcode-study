var findInMountainArray = function(target, arr) {
const n = arr.length;
let left = 0;
let right = n - 1;
while (left < right) {
const mid = Math.floor((left + right) / 2);
if (arr[mid] < arr[mid + 1]) {
left = mid + 1;
} else {
right = mid;
}
}
const peak = left;
let res = search(arr, target, 0, peak, true);
if (res !== -1) {
return res;
}
return search(arr, target, peak + 1, n - 1, false);
};
function search(arr, target, left, right, ascending) {
while (left <= right) {
const mid = Math.floor((left + right) / 2);
if (arr[mid] === target) {
return mid;
}
if (ascending) {
if (arr[mid] < target) left = mid + 1;
else right = mid - 1;
} else {
if (arr[mid] > target) left = mid + 1;
else right = mid - 1;
}
}
return -1;
}

How It Works

PHASE 1 find the peak: compare arr[mid] with arr[mid + 1]
PHASE 2 binary search the ascending side [0, peak]
PHASE 3 binary search the descending side [peak + 1, n - 1]
  1. A mountain array goes strictly up, then strictly down.
  2. Find the peak index where the slope flips from up to down.
  3. Search the ascending half first (normal binary search).
  4. If the target is there, return that index (the smallest match).
  5. Otherwise search the descending half (comparison flips).
