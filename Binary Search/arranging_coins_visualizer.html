441. Arranging Coins

Binary search on the number of rows

Live Debugger

var arrangeCoins = function(n) {
let left = 1, right = n, res = 0;
while (left <= right) {
const mid = Math.floor((left + right) / 2);
const coins = (mid * (mid + 1)) / 2;
if (coins > n) {
right = mid - 1;
} else {
res = mid;
left = mid + 1;
}
}
return res;
};
