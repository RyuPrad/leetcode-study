class TimeMap {
constructor() {
this.store = new Map();
}
set(key, value, timestamp) {
if (!this.store.has(key)) {
this.store.set(key, []);
}
this.store.get(key).push([timestamp, value]);
}
get(key, timestamp) {
const arr = this.store.get(key) || [];
let left = 0;
let right = arr.length - 1;
let res = "";
while (left <= right) {
const mid = Math.floor((left + right) / 2);
if (arr[mid][0] <= timestamp) {
res = arr[mid][1];
left = mid + 1;
} else {
right = mid - 1;
}
}
return res;
}
}

Operation Trace

How It Works

set(key, value, timestamp) appends [timestamp, value] to that key's list.
Timestamps arrive in increasing order, so each list stays sorted.
get(key, timestamp) binary-searches for the largest timestamp <= the query.
  1. The store maps each key to a time-ordered list of [timestamp, value].
  2. set just pushes onto the end of the list (already sorted).
  3. get binary-searches the timestamps with left, right, mid.
  4. When arr[mid][0] <= timestamp, record the value and move right.
  5. Otherwise move left. The last recorded value wins, or "" if none.
