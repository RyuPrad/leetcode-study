926. Flip String to Monotone Increasing

Flip this 0, or flip every earlier 1

Live Debugger

var minFlipsMonoIncr = function(s) {
let ones = 0, res = 0;
for (const ch of s) {
if (ch === '1') {
ones++;
} else {
res = Math.min(res + 1, ones);
}
}
return res;
};
