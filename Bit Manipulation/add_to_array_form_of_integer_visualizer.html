989. Add to Array-Form of Integer

Add from the right with k as the carry

Live Debugger

var addToArrayForm = function(num, k) {
const res = [];
let i = num.length - 1;
while (i >= 0 || k > 0) {
if (i >= 0) {
k += num[i];
}
res.push(k % 10);
k = Math.floor(k / 10);
i--;
}
return res.reverse();
};
