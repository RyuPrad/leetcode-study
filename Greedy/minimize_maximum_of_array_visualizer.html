2439. Minimize Maximum of Array

Each prefix can at best share its total evenly

Live Debugger

var minimizeArrayValue = function(nums) {
let prefix = 0, res = 0;
for (let i = 0; i < nums.length; i++) {
prefix += nums[i];
res = Math.max(res, Math.ceil(prefix / (i + 1)));
}
return res;
};
