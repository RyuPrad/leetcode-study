var subsetXORSum = function(nums) {
let ans = 0;
function dfs(i, xorSoFar) {
if (i === nums.length) {
ans += xorSoFar;
return;
}
dfs(i + 1, xorSoFar ^ nums[i]);
dfs(i + 1, xorSoFar);
}
dfs(0, 0);
return ans;
};
