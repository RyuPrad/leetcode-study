var letterCombinations = function(digits) {
if (digits.length === 0) return [];
const map = { '2':'abc','3':'def','4':'ghi','5':'jkl','6':'mno','7':'pqrs','8':'tuv','9':'wxyz' };
const res = [];
const path = [];
function dfs(i) {
if (i === digits.length) {
res.push(path.join(''));
return;
}
for (const c of map[digits[i]]) {
path.push(c);
dfs(i + 1);
path.pop();
}
}
dfs(0);
return res;
};
