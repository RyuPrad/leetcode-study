var totalNQueens = function(n) {
let count = 0;
const cols = new Set(); const diag = new Set(); const antiDiag = new Set();
function dfs(r) {
if (r === n) { count++; return; }
for (let c = 0; c < n; c++) {
if (cols.has(c) || diag.has(r - c) || antiDiag.has(r + c)) continue;
cols.add(c); diag.add(r - c); antiDiag.add(r + c);
dfs(r + 1);
cols.delete(c); diag.delete(r - c); antiDiag.delete(r + c);
}
}
dfs(0);
return count;
};
