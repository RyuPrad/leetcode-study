1514. Path with Maximum Probability

Dijkstra with a max-heap of probabilities

var maxProbability = function(n, edges, succProb, start_node, end_node) {
const adj = Array.from({ length: n }, () => []);
for (let i = 0; i < edges.length; i++) {
const [a, b] = edges[i];
adj[a].push([b, succProb[i]]);
adj[b].push([a, succProb[i]]);
}
const best = new Array(n).fill(0);
best[start_node] = 1;
const heap = new Heap((x, y) => x[0] > y[0]);
heap.push([1, start_node]);
while (heap.size()) {
const [prob, node] = heap.pop();
if (node === end_node) {
return prob;
}
if (prob < best[node]) {
continue;
}
for (const [next, edgeProb] of adj[node]) {
const candidate = prob * edgeProb;
if (candidate > best[next]) {
best[next] = candidate;
heap.push([candidate, next]);
}
}
}
return 0;
};
