var carPooling = function(trips, capacity) {
trips.sort((a, b) => a[1] - b[1]);
const heap = new MinHeap(); // by end el[0]
let cur = 0;
for (const [num, start, end] of trips) {
while (heap.size() && heap.peek()[0] <= start) {
cur -= heap.pop()[1];
}
cur += num;
if (cur > capacity) return false;
heap.push([end, num]);
}
return true;
};
