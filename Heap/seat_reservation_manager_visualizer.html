1845. Seat Reservation Manager

A counter for new seats, a heap for returned ones

Live Debugger

class SeatManager {
constructor(n) {
this.next = 1;
this.freed = new Heap((a, b) => a < b);
}
reserve() {
if (this.freed.size()) {
return this.freed.pop();
}
return this.next++;
}
unreserve(seatNumber) {
this.freed.push(seatNumber);
}
}
