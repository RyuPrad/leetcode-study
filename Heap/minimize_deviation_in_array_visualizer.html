1675. Minimize Deviation in Array

Start every value at its maximum, then shrink the largest

Live Debugger

var minimumDeviation = function(nums) {
const heap = new Heap((a, b) => a > b);
let low = Infinity;
for (const num of nums) {
const value = num % 2 ? num * 2 : num;
heap.push(value);
low = Math.min(low, value);
}
let res = Infinity;
while (true) {
const high = heap.pop();
res = Math.min(res, high - low);
if (high % 2) {
break;
}
heap.push(high / 2);
low = Math.min(low, high / 2);
}
return res;
};
