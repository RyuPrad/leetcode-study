1383. Maximum Performance of a Team

Fix the lowest efficiency, keep the fastest k

var maxPerformance = function(n, speed, efficiency, k) {
const engineers = speed.map((s, i) => [efficiency[i], s]).sort((a, b) => b[0] - a[0]);
const heap = new Heap((a, b) => a < b);
let sum = 0, best = 0n;
for (const [eff, spd] of engineers) {
heap.push(spd);
sum += spd;
if (heap.size() > k) {
sum -= heap.pop();
}
const performance = BigInt(sum) * BigInt(eff);
if (performance > best) {
best = performance;
}
}
return Number(best % 1000000007n);
};
