Leetcode 1046. Last Stone Weight

Max-Heap (smash the two heaviest stones)

var lastStoneWeight = function(stones) {
const heap = new MaxHeap();
for (const s of stones) {
heap.push(s);
}
while (heap.size() > 1) {
const a = heap.pop();
const b = heap.pop();
if (a !== b) {
heap.push(a - b);
}
}
return heap.size() ? heap.peek() : 0;
};
