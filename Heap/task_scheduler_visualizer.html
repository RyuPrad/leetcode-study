var leastInterval = function(tasks, n) {
const freq = {};
for (const t of tasks) freq[t] = (freq[t] || 0) + 1;
const heap = new MaxHeap();
for (const key in freq) heap.push(freq[key]);
let time = 0;
const queue = [];
while (heap.size() || queue.length) {
time++;
if (heap.size()) {
const cnt = heap.pop() - 1;
if (cnt > 0) queue.push([cnt, time + n]);
}
if (queue.length && queue[0][1] === time) {
heap.push(queue.shift()[0]);
}
}
return time;
};
