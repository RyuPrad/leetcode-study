1882. Process Tasks Using Servers

Idle servers by weight, busy servers by finish time

Live Debugger

var assignTasks = function(servers, tasks) {
const free = new Heap((a, b) => a[0] < b[0] || (a[0] === b[0] && a[1] < b[1]));
const busy = new Heap((a, b) => a[0] < b[0]);
for (let i = 0; i < servers.length; i++) {
free.push([servers[i], i]);
}
const res = [];
let time = 0;
for (let j = 0; j < tasks.length; j++) {
time = Math.max(time, j);
if (!free.size()) {
time = busy.peek()[0];
}
while (busy.size() && busy.peek()[0] <= time) {
const [, weight, index] = busy.pop();
free.push([weight, index]);
}
const [weight, index] = free.pop();
res.push(index);
busy.push([time + tasks[j], weight, index]);
}
return res;
};
