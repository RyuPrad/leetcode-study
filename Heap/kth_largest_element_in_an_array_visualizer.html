Leetcode 215. Kth Largest Element in an Array

Min-Heap capped at size k (root = kth largest)

Live Debugger

var findKthLargest = function(nums, k) {
const heap = new MinHeap();
for (const num of nums) {
heap.push(num);
if (heap.size() > k) {
heap.pop();
}
}
return heap.peek();
};
