Alternating smallest and largest

var rearrangeArray = function(nums) {
nums.sort((a, b) => a - b);
const res = [];
let left = 0, right = nums.length - 1;
while (res.length < nums.length) {
res.push(nums[left]);
left++;
if (left <= right) {
res.push(nums[right]);
right--;
}
}
return res;
};
