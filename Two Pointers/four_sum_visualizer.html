18. 4Sum

Given an array nums and a target, return all unique quadruples [nums[i], nums[j], nums[left], nums[right]] that sum to target. Sort first, fix i and j, then converge left and right while skipping duplicates.

var fourSum = function(nums, target) {
nums.sort((a, b) => a - b);
const res = [];
const n = nums.length;
for (let i = 0; i < n - 3; i++) {
if (i > 0 && nums[i] === nums[i - 1]) continue;
for (let j = i + 1; j < n - 2; j++) {
if (j > i + 1 && nums[j] === nums[j - 1]) continue;
let left = j + 1;
let right = n - 1;
while (left < right) {
const sum = nums[i] + nums[j] + nums[left] + nums[right];
if (sum === target) {
res.push([nums[i], nums[j], nums[left], nums[right]]);
left++;
right--;
while (left < right && nums[left] === nums[left - 1]) left++;
while (left < right && nums[right] === nums[right + 1]) right--;
} else if (sum < target) {
left++;
} else {
right--;
}
}
}
}
return res;
};

Step Trace

How It Works

  1. Sort nums so duplicates sit together and pointers move logically.
  2. Fix the first two numbers with i and j, skipping duplicate fixed values.
  3. Use left and right to find the other two numbers.
  4. If sum < target, move left right; if sum > target, move right left.
  5. If sum === target, save the quadruple, move both pointers, then skip duplicates.
