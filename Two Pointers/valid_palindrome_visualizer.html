var isPalindrome = function(s) {
let left = 0;
let right = s.length - 1;
while (left < right) {
while (left < right && !isAlphaNumeric(s[left])) left++;
while (left < right && !isAlphaNumeric(s[right])) right--;
if (s[left].toLowerCase() !== s[right].toLowerCase()) {
return false;
}
left++;
right--;
}
return true;
};
function isAlphaNumeric(char) {
return (char >= 'a' && char <= 'z') ||
(char >= 'A' && char <= 'Z') ||
(char >= '0' && char <= '9');
}

How It Works

  1. Use two pointers: left at the start and right at the end.
  2. Skip characters that are not letters or numbers.
  3. Compare valid characters in lowercase.
  4. If a mismatch appears, immediately return false.
  5. If the pointers meet or cross, return true.
