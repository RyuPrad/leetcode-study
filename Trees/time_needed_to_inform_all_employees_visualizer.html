1376. Time Needed to Inform All Employees

Walk the management tree from the head, adding each manager's time

Live Debugger

var numOfMinutes = function(n, headID, manager, informTime) {
const reports = Array.from({ length: n }, () => []);
for (let i = 0; i < n; i++) {
if (manager[i] !== -1) {
reports[manager[i]].push(i);
}
}
const queue = [[headID, 0]];
let res = 0;
for (let head = 0; head < queue.length; head++) {
const [id, time] = queue[head];
res = Math.max(res, time);
for (const next of reports[id]) {
queue.push([next, time + informTime[id]]);
}
}
return res;
};
