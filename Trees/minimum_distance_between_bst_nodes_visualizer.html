783. Minimum Distance Between BST Nodes

In-order visits sorted values; compare neighbours

Live Debugger

var minDiffInBST = function(root) {
const stack = [];
let node = root, prev = null, res = Infinity;
while (node || stack.length) {
while (node) {
stack.push(node);
node = node.left;
}
node = stack.pop();
if (prev !== null) {
res = Math.min(res, node.val - prev);
}
prev = node.val;
node = node.right;
}
return res;
};
