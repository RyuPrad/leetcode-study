513. Find Bottom Left Tree Value

Breadth-first, right child first: the last node is the bottom-left

Live Debugger

var findBottomLeftValue = function(root) {
const queue = [root];
let node = null;
for (let head = 0; head < queue.length; head++) {
node = queue[head];
if (node.right) {
queue.push(node.right);
}
if (node.left) {
queue.push(node.left);
}
}
return node.val;
};
