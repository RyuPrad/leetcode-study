var serialize = function(root) {
const res = [];
function dfs(node) {
if (!node) { res.push("null"); return; }
res.push(node.val);
dfs(node.left);
dfs(node.right);
}
dfs(root);
return res.join(",");
};
var deserialize = function(data) {
const vals = data.split(",");
let i = 0;
function dfs() {
if (vals[i] === "null") { i++; return null; }
const node = new TreeNode(Number(vals[i]));
i++;
node.left = dfs();
node.right = dfs();
return node;
}
return dfs();
};
