538. Convert BST to Greater Tree

Visit from largest to smallest, carrying a running sum

Live Debugger

var convertBST = function(root) {
const stack = [];
let node = root, sum = 0;
while (node || stack.length) {
while (node) {
stack.push(node);
node = node.right;
}
node = stack.pop();
sum += node.val;
node.val = sum;
node = node.left;
}
return root;
};
