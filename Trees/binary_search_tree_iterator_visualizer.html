173. Binary Search Tree Iterator

A stack of left spines: its top is always the next smallest value

Live Debugger

class BSTIterator {
constructor(root) {
this.stack = [];
this.pushLeft(root);
}
pushLeft(node) {
while (node) {
this.stack.push(node);
node = node.left;
}
}
next() {
const node = this.stack.pop();
this.pushLeft(node.right);
return node.val;
}
hasNext() {
return this.stack.length > 0;
}
}
