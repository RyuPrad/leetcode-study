1721. Swapping Nodes in a Linked List

A k - 1 node gap finds the kth node from the end

Live Debugger

var swapNodes = function(head, k) {
let fast = head;
for (let i = 1; i < k; i++) {
fast = fast.next;
}
const first = fast;
let slow = head;
while (fast.next) {
slow = slow.next;
fast = fast.next;
}
[first.val, slow.val] = [slow.val, first.val];
return head;
};
