725. Split Linked List in Parts

Every part gets length / k nodes; the first length % k get one more

Live Debugger

var splitListToParts = function(head, k) {
let length = 0;
for (let node = head; node; node = node.next) {
length++;
}
const size = Math.floor(length / k), extra = length % k;
const res = new Array(k).fill(null);
let curr = head;
for (let i = 0; i < k && curr; i++) {
res[i] = curr;
const partSize = size + (i < extra ? 1 : 0);
for (let j = 1; j < partSize; j++) {
curr = curr.next;
}
const next = curr.next;
curr.next = null;
curr = next;
}
return res;
};
