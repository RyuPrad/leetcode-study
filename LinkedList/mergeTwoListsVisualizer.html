var mergeTwoLists = function(list1, list2) {
const dummy = new ListNode();
let tail = dummy;
while (list1 !== null && list2 !== null) {
if (list1.val <= list2.val) {
tail.next = list1;
list1 = list1.next;
} else {
tail.next = list2;
list2 = list2.next;
}
tail = tail.next;
}
// One is null; attach what's left
tail.next = list1 !== null ? list1 : list2;
return dummy.next;
};
