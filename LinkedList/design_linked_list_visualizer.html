707. Design Linked List

A dummy node and a size: every position has a node before it

Live Debugger

class MyLinkedList {
constructor() {
this.dummy = new ListNode(0);
this.size = 0;
}
get(index) {
if (index < 0 || index >= this.size) {
return -1;
}
let curr = this.dummy.next;
for (let i = 0; i < index; i++) {
curr = curr.next;
}
return curr.val;
}
addAtHead(val) {
this.addAtIndex(0, val);
}
addAtTail(val) {
this.addAtIndex(this.size, val);
}
addAtIndex(index, val) {
if (index > this.size) {
return;
}
let prev = this.dummy;
for (let i = 0; i < index; i++) {
prev = prev.next;
}
prev.next = new ListNode(val, prev.next);
this.size++;
}
deleteAtIndex(index) {
if (index < 0 || index >= this.size) {
return;
}
let prev = this.dummy;
for (let i = 0; i < index; i++) {
prev = prev.next;
}
prev.next = prev.next.next;
this.size--;
}
}
