var reverseBetween = function(head, left, right) {
const dummy = new ListNode(0, head);
let prev = dummy;
for (let i = 0; i < left - 1; i++) {
prev = prev.next;
}
let curr = prev.next;
for (let i = 0; i < right - left; i++) {
const next = curr.next;
curr.next = next.next;
next.next = prev.next;
prev.next = next;
}
return dummy.next;
};
