160. Intersection of Two Linked Lists

Walk both lists, then switch: both pointers cover the same distance

Live Debugger

var getIntersectionNode = function(headA, headB) {
let a = headA, b = headB;
while (a !== b) {
a = a ? a.next : headB;
b = b ? b.next : headA;
}
return a;
};
