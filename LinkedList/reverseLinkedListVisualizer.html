LeetCode 206 — Reverse Linked List

Goal: reverse the linked list in-place by flipping each node's next pointer. The key pattern is: save the old next node, reverse the arrow, then move both pointers forward.

var reverseList = function(head) {
let prev = null;
let curr = head;
while (curr !== null) {
let next = curr.next;
curr.next = prev;
prev = curr;
curr = next;
}
return prev;
};
