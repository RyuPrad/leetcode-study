203. Remove Linked List Elements

A dummy node, then skip every matching next node

Live Debugger

var removeElements = function(head, val) {
const dummy = new ListNode(0, head);
let curr = dummy;
while (curr.next) {
if (curr.next.val === val) {
curr.next = curr.next.next;
} else {
curr = curr.next;
}
}
return dummy.next;
};
