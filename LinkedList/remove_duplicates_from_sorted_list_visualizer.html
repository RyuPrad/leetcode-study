83. Remove Duplicates from Sorted List

Skip the next node while it repeats the current value

Live Debugger

var deleteDuplicates = function(head) {
let curr = head;
while (curr && curr.next) {
if (curr.val === curr.next.val) {
curr.next = curr.next.next;
} else {
curr = curr.next;
}
}
return head;
};
