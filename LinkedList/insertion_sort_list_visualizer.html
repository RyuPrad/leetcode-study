147. Insertion Sort List

Insert each node after the last smaller sorted node

Live Debugger

var insertionSortList = function(head) {
const dummy = new ListNode(0);
let curr = head;
while (curr) {
const next = curr.next;
let prev = dummy;
while (prev.next && prev.next.val < curr.val) {
prev = prev.next;
}
curr.next = prev.next;
prev.next = curr;
curr = next;
}
return dummy.next;
};
