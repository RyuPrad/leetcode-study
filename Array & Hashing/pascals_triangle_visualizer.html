118. Pascal's Triangle

Each entry adds the two above it

var generate = function(numRows) {
const res = [[1]];
for (let i = 1; i < numRows; i++) {
const prev = res[i - 1];
const row = [1];
for (let j = 1; j < i; j++) {
row.push(prev[j - 1] + prev[j]);
}
row.push(1);
res.push(row);
}
return res;
};
