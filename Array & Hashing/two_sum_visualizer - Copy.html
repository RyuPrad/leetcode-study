function twoSum(nums, target) {
const seen = new Map();
for (let i = 0; i < nums.length; i++) {
const current = nums[i];
const needed = target - current;
if (seen.has(needed)) {
return [seen.get(needed), i];
}
seen.set(current, i);
}
}

Interview Approach

  1. Create a hash map named seen.
  2. For each number, compute needed = target - current.
  3. If needed already exists in the map, return the old index and current index.
  4. Otherwise, store current → i.

Complexity

  • Time: O(n)
