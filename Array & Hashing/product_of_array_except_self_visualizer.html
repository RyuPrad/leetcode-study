Leetcode 238. Product of Array Except Self Visualizer

Given an integer array nums, return an array answer
where answer[i] equals the product of every element except nums[i].
Goal Solve in O(n) time, without division, using prefix and suffix products.

var productExceptSelf = function(nums) {
const n = nums.length;
const res = new Array(n).fill(1);
let prefix = 1;
for (let i = 0; i < n; i++) {
res[i] = prefix;
prefix *= nums[i];
}
let suffix = 1;
for (let i = n - 1; i >= 0; i--) {
res[i] *= suffix;
suffix *= nums[i];
}
return res;
};
