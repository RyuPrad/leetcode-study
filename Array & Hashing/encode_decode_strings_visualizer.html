class Codec {
encode(strs) {
let encoded = "";
for (let str of strs) {
encoded += str.length + "#" + str;
}
return encoded;
}
decode(s) {
const decoded = [];
let i = 0;
while (i < s.length) {
let j = i;
while (s[j] !== "#") j++;
const len = Number(s.slice(i, j));
const start = j + 1;
decoded.push(s.slice(start, start + len));
i = start + len;
}
return decoded;
}
}
