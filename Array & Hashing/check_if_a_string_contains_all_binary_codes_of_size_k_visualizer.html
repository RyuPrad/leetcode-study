1461. Check If a String Contains All Binary Codes of Size K

Collecting every window of length k

Live Debugger

var hasAllCodes = function(s, k) {
const codes = new Set();
for (let i = 0; i + k <= s.length; i++) {
codes.add(s.slice(i, i + k));
}
return codes.size === 1 << k;
};
