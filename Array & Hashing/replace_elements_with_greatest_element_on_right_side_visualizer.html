1299. Replace Elements with Greatest Element on Right Side

Carrying the maximum leftward

Live Debugger

var replaceElements = function(arr) {
let rightMax = -1;
for (let i = arr.length - 1; i >= 0; i--) {
const current = arr[i];
arr[i] = rightMax;
rightMax = Math.max(rightMax, current);
}
return arr;
};
