2405. Optimal Partition of String

Starting a new piece at the first repeat

var partitionString = function(s) {
let seen = new Set(), res = 1;
for (const c of s) {
if (seen.has(c)) {
res++;
seen = new Set();
}
seen.add(c);
}
return res;
};
