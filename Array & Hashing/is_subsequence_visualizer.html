392. Is Subsequence

Matching s inside t, left to right

Live Debugger

var isSubsequence = function(s, t) {
let i = 0;
for (let j = 0; j < t.length && i < s.length; j++) {
if (s[i] === t[j]) {
i++;
}
}
return i === s.length;
};
