const count = new Map();
for (const num of nums) {

}
const buckets = Array.from({ length: nums.length + 1 }, () => []);
for (const [num, freq] of count) {
buckets[freq].push(num);
}
const res = [];
for (let freq = buckets.length - 1; freq >= 0; freq--) {
for (const num of buckets[freq]) {

}
