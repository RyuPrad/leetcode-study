1963. Minimum Number of Swaps to Make the String Balanced

The worst prefix decides

Live Debugger

var minSwaps = function(s) {
let extraClose = 0, maxClose = 0;
for (const c of s) {
if (c === ']') {
extraClose++;
} else {
extraClose--;
}
maxClose = Math.max(maxClose, extraClose);
}
return Math.floor((maxClose + 1) / 2);
};
