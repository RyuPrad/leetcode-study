1189. Maximum Number of Balloons

The scarcest letter decides

var maxNumberOfBalloons = function(text) {
const count = new Map();
for (const c of text) {
count.set(c, (count.get(c) || 0) + 1);
}
const need = new Map([['b', 1], ['a', 1], ['l', 2], ['o', 2], ['n', 1]]);
let res = Infinity;
for (const c of need.keys()) {
res = Math.min(res, Math.floor((count.get(c) || 0) / need.get(c)));
}
return res;
};
