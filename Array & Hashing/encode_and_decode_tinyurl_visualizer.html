535. Encode and Decode TinyURL

The list index is the short code

Live Debugger

class Codec {
constructor() {
this.urls = [];
}
encode(longUrl) {
this.urls.push(longUrl);
return 'http://tinyurl.com/' + (this.urls.length - 1);
}
decode(shortUrl) {
const id = Number(shortUrl.slice(shortUrl.lastIndexOf('/') + 1));
return this.urls[id];
}
}
