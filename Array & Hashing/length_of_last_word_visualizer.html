58. Length of Last Word

Reading the last word from the end

Live Debugger

var lengthOfLastWord = function(s) {
let i = s.length - 1, length = 0;
while (i >= 0 && s[i] === ' ') {
i--;
}
while (i >= 0 && s[i] !== ' ') {
length++;
i--;
}
return length;
};
