function getConcatenation(nums) {
const res = new Array(nums.length * 2);
for (let i = 0; i < nums.length; i++) {
res[i] = nums[i];
res[i + nums.length] = nums[i];
}
return res;
}

Interview Approach

  1. Create res with size nums.length * 2.
  2. Loop through every index i in nums.
  3. Copy nums[i] into the first half: res[i].
  4. Copy the same value into the second half: res[i + nums.length].
  5. Return the completed res array.
