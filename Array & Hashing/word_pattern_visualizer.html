290. Word Pattern

Letters and words must match one to one

Live Debugger

var wordPattern = function(pattern, s) {
const words = s.split(' ');
if (words.length !== pattern.length) {
return false;
}
const charToWord = new Map(), wordToChar = new Map();
for (let i = 0; i < pattern.length; i++) {
const c = pattern[i], word = words[i];
if ((charToWord.has(c) && charToWord.get(c) !== word) || (wordToChar.has(word) && wordToChar.get(word) !== c)) {
return false;
}
charToWord.set(c, word);
wordToChar.set(word, c);
}
return true;
};
