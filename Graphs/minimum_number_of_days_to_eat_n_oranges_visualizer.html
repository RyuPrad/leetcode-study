1553. Minimum Number of Days to Eat N Oranges

Only ever eat single oranges to reach a multiple of 2 or 3

Live Debugger

var minDays = function(n) {
const memo = new Map([[0, 0], [1, 1]]);
const dfs = k => {
if (memo.has(k)) {
return memo.get(k);
}
const half = k % 2 + dfs(Math.floor(k / 2));
const third = k % 3 + dfs(Math.floor(k / 3));
const res = 1 + Math.min(half, third);
memo.set(k, res);
return res;
};
return dfs(n);
};
