Leetcode 200. Number of Islands

Separate rows with ; and cells with commas, for example 1,1,0; 0,1,0. Use 0 or 1, with equal row lengths.

function numIslands(grid) {
const rows = grid.length;
const cols = grid[0].length;
let count = 0;
const dfs = (r, c) => {
if (r < 0 || r >= rows || c < 0 || c >= cols || grid[r][c] !== "1") return;
grid[r][c] = "0";
dfs(r + 1, c);
dfs(r - 1, c);
dfs(r, c + 1);
dfs(r, c - 1);
};
for (let r = 0; r < rows; r++) {
for (let c = 0; c < cols; c++) {
if (grid[r][c] === "1") {
count++;
dfs(r, c);
}
}
}
return count;
}
