var findSmallestSetOfVertices = function(n, edges) {
const hasIncoming = new Array(n).fill(false);
for (const [, to] of edges) {
hasIncoming[to] = true;
}
const res = [];
for (let node = 0; node < n; node++) {
if (!hasIncoming[node]) {
res.push(node);
}
}
return res;
};
