691. Stickers to Spell Word

Cover the first missing letter, remember every remainder

var minStickers = function(stickers, target) {
const memo = new Map([['', 0]]);
const solve = rest => {
if (memo.has(rest)) {
return memo.get(rest);
}
let best = Infinity;
for (const sticker of stickers) {
if (!sticker.includes(rest[0])) {
continue;
}
let next = rest;
for (const ch of sticker) {
next = next.replace(ch, '');
}
best = Math.min(best, 1 + solve(next));
}
memo.set(rest, best);
return best;
};
const res = solve([...target].sort().join(''));
return res === Infinity ? -1 : res;
};
