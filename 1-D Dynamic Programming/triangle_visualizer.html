Fold the triangle from the bottom up

var minimumTotal = function(triangle) {
const dp = triangle[triangle.length - 1].slice();
for (let r = triangle.length - 2; r >= 0; r--) {
for (let c = 0; c <= r; c++) {
dp[c] = triangle[r][c] + Math.min(dp[c], dp[c + 1]);
}
}
return dp[0];
};
