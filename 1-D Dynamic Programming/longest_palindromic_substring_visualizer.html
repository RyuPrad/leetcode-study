function longestPalindrome(s) {
if (s.length < 2) return s;
let start = 0, maxLen = 1;
const expand = (l, r) => {
while (l >= 0 && r < s.length && s[l] === s[r]) {
if (r - l + 1 > maxLen) { start = l; maxLen = r - l + 1; }
l--;
r++;
}
};
for (let i = 0; i < s.length; i++) {
expand(i, i);
expand(i, i + 1);
}
return s.substring(start, start + maxLen);
}
