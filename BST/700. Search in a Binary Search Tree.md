
```js
/**
 * @param {TreeNode} root
 * @param {number} val
 * @return {TreeNode}
 */
var searchBST = function(root, val) {
    let curr = root;

    // Keep walking as long as there are nodes to look at
    while (curr !== null) {
        // 1. Found it!
        if (curr.val === val) {
            return curr;
        }

        // 2. Is the target smaller? Go Left.
        if (val < curr.val) {
            curr = curr.left;
        } 
        // 3. Is the target larger? Go Right.
        else {
            curr = curr.right;
        }
    }

    // 4. We hit null, meaning we searched the whole path and it's not there.
    return null;
};
```

## Variable Pattern

| Name | Meaning |
|---|---|
| root | tree root |
| curr | current node while searching |
| val | target value |
