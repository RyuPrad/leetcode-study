function minInterval(intervals, queries) {
intervals.sort((a, b) => a[0] - b[0]);
const sortedQueries = queries.map((q, idx) => [q, idx]).sort((a, b) => a[0] - b[0]);
const res = new Array(queries.length).fill(-1);
const heap = [];
let i = 0;
for (const [q, idx] of sortedQueries) {
while (i < intervals.length && intervals[i][0] <= q) {
const l = intervals[i][0], r = intervals[i][1];
heap.push([r - l + 1, r]);
heap.sort((a, b) => a[0] - b[0]);
i++;
}
while (heap.length > 0 && heap[0][1] < q) {
heap.shift();
}
res[idx] = heap.length > 0 ? heap[0][0] : -1;
}
return res;
}
