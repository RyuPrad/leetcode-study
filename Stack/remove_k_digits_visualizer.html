402. Remove K Digits

A nondecreasing stack of digits

Live Debugger

var removeKdigits = function(num, k) {
const stack = [];
for (const digit of num) {
while (k > 0 && stack.length > 0 && stack[stack.length - 1] > digit) {
stack.pop();
k--;
}
stack.push(digit);
}
while (k > 0) {
stack.pop();
k--;
}
const res = stack.join('').replace(/^0+/, '');
return res === '' ? '0' : res;
};
