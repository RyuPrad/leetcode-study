2390. Removing Stars From a String

Each star pops the closest letter

Live Debugger

var removeStars = function(s) {
const stack = [];
for (const c of s) {
if (c === '*') {
stack.pop();
} else {
stack.push(c);
}
}
return stack.join('');
};
