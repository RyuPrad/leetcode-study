class MinStack {
constructor() {
this.stack = [];
}
push(val) {
const min = this.stack.length === 0
? val : Math.min(val, this.getMin());
this.stack.push([val, min]);
}
pop() {
this.stack.pop();
}
top() {
return this.stack[this.stack.length - 1][0];
}
getMin() {
return this.stack[this.stack.length - 1][1];
}
}
