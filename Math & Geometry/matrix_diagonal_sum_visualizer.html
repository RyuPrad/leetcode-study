1572. Matrix Diagonal Sum

Two diagonal cells per row

var diagonalSum = function(mat) {
const n = mat.length;
let res = 0;
for (let i = 0; i < n; i++) {
res += mat[i][i];
if (i !== n - 1 - i) {
res += mat[i][n - 1 - i];
}
}
return res;
};
