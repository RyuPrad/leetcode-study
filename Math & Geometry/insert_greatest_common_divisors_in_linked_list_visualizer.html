class ListNode {
constructor(val, next = null) { this.val = val; this.next = next; }
}
function insertGreatestCommonDivisors(head) {
const gcd = (a, b) => (b === 0 ? a : gcd(b, a % b));
let curr = head;
while (curr.next !== null) {
const node = new ListNode(gcd(curr.val, curr.next.val));
node.next = curr.next;
curr.next = node;
curr = node.next;
}
return head;
}
