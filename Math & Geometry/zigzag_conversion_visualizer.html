6. Zigzag Conversion

Each character joins one row

Live Debugger

var convert = function(s, numRows) {
if (numRows === 1) {
return s;
}
const rows = new Array(numRows).fill('');
let row = 0, step = 1;
for (const c of s) {
rows[row] += c;
if (row === 0) {
step = 1;
} else if (row === numRows - 1) {
step = -1;
}
row += step;
}
return rows.join('');
};
